


void discard_preprocessor_directives(const char *input_file,
                                     const char *output_file)
{
    FILE *in = fopen(input_file, "r");
    FILE *out = fopen(output_file, "w");

    if (!in || !out) {
        perror("File error");
        exit(EXIT_FAILURE);
    }

    int ch, next;
    int at_line_start = 1;
    int in_directive = 0;
    int in_string = 0, in_char = 0;
    int in_comment = 0;
    int line_continuation = 0;

    while ((ch = fgetc(in)) != EOF) {

        
        if (!in_comment && ch == '"' && !in_char) {
            in_string = !in_string;
            if (!in_directive) fputc(ch, out);
            at_line_start = 0;
            continue;
        }

        
        if (!in_comment && ch == '\'' && !in_string) {
            in_char = !in_char;
            if (!in_directive) fputc(ch, out);
            at_line_start = 0;
            continue;
        }

        /* ---------- Handle comments ---------- */
        if (!in_string && !in_char && !in_comment && ch == '/') {
            next = fgetc(in);

            /* Single-line comment */
            if (next == '/') {
                while ((ch = fgetc(in)) != EOF && ch != '\n');
                if (!in_directive && ch == '\n') fputc('\n', out);
                at_line_start = 1;
                continue;
            }

            /* Multi-line comment */
            if (next == '*') {
                while ((ch = fgetc(in)) != EOF) {
                    if (ch == '*' && (next = fgetc(in)) == '/')
                        break;
                }
                continue;
            }

            ungetc(next, in);
        }

        /* ---------- Detect preprocessor directive ---------- */
        if (!in_string && !in_char && !in_comment && at_line_start) {
            if (ch == '#')
                in_directive = 1;
            else if (ch != ' ' && ch != '\t')
                at_line_start = 0;
        }

        /* ---------- Handle line continuation ---------- */
        if (ch == '\\' && in_directive)
            line_continuation = 1;

        if (ch == '\n') {
            at_line_start = 1;
            if (!line_continuation)
                in_directive = 0;
            line_continuation = 0;
        } else {
            at_line_start = 0;
        }

        /* ---------- Output ---------- */
        if (!in_directive)
            fputc(ch, out);
    }

    fclose(in);
    fclose(out);
}

int main(int argc, char *argv[])
{
    if (argc != 3) {
        fprintf(stderr,
                "Usage: %s <input_c_file> <output_c_file>\n",
                argv[0]);
        return EXIT_FAILURE;
    }

    discard_preprocessor_directives(argv[1], argv[2]);
    return EXIT_SUCCESS;
}
